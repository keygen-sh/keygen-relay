# add a license so serve can start cleanly
exec sh -c 'key=$(cat "$1"); pub=$(cat "$2"); relay add --file "$3" --key "$key" --public-key "$pub"' _ $E2E_EXPORT_DIR/license.key $E2E_EXPORT_DIR/public.key $E2E_EXPORT_DIR/license.lic

env PORT=63242

# hold port with first process
exec relay serve --port $PORT --no-heartbeats &relay_chaos_port_holder&
exec sleep 1

# second process on same port should emit startup diagnostics
exec relay serve --port $PORT --no-heartbeats
stderr 'address already in use'

kill relay_chaos_port_holder
