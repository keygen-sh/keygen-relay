# add one single-seat license to a persisted db
env DB=recovery-restart.sqlite
env PORT=63230
exec sh -c 'key=$(cat "$1"); pub=$(cat "$2"); relay add --database "$3" --file "$4" --key "$key" --public-key "$pub" --seats 1' _ $E2E_EXPORT_DIR/license.key $E2E_EXPORT_DIR/public.key $DB $E2E_EXPORT_DIR/license.lic

# first server process claims seat
exec relay serve --database $DB --port $PORT --no-heartbeats &relay_recovery_1&
exec sleep 1
exec curl -s -o /dev/null -w "%{http_code}" -X PUT http://localhost:$PORT/v1/nodes/recovery-node
stdout '201'
kill relay_recovery_1

# restart with same db: claim state should persist
exec relay serve --database $DB --port $PORT --no-heartbeats &relay_recovery_2&
exec sleep 1
exec curl -s -o /dev/null -w "%{http_code}" -X PUT http://localhost:$PORT/v1/nodes/recovery-node
stdout '409'
exec curl -s -o /dev/null -w "%{http_code}" -X DELETE http://localhost:$PORT/v1/nodes/recovery-node
stdout '204'
exec curl -s -o /dev/null -w "%{http_code}" -X PUT http://localhost:$PORT/v1/nodes/recovery-node-2
stdout '201'
kill relay_recovery_2
